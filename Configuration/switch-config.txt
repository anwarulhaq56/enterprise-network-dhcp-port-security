! Project 02 - Enterprise Network with DHCP & Port Security
! Switch: SW1
! Function: VLAN Segmentation + Port Security

enable
configure terminal

! VLAN Configuration
vlan 10
 name HR
exit

vlan 20
 name IT
exit

vlan 30
 name FINANCE
exit

! HR Access Ports
interface range fa0/2-3
 switchport mode access
 switchport access vlan 10
exit

! IT Access Ports
interface range fa0/4-5
 switchport mode access
 switchport access vlan 20
exit

! Finance Access Ports
interface range fa0/6-7
 switchport mode access
 switchport access vlan 30
exit

! 802.1Q Trunk
interface fa0/1
 switchport mode trunk
exit

! Port Security - Fa0/2
interface fa0/2
 switchport mode access
 switchport port-security
 switchport port-security mac-address 00d0.d344.6cb1
 switchport port-security maximum 1
 switchport port-security violation shutdown
exit

! Port Security - Fa0/3
interface fa0/3
 switchport mode access
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
exit

! Port Security - Fa0/4 to Fa0/7
interface range fa0/4-7
 switchport mode access
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
exit

end
